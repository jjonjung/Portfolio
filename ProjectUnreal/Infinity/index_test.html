// EnemyCharacter.h
class AEnemyCharacter : public ACharacter
{
    UPROPERTY()
    UMovementComponent* MovementComp;
    
    UPROPERTY()
    UAttackComponent* AttackComp;
    
    UPROPERTY()
    UPerceptionComponent* PerceptionComp;
    
    UPROPERTY()
    UHealthComponent* HealthComp;
};

void AEnemyCharacter::Tick(float DeltaTime)
{
    // 각 컴포넌트가 독립적으로 동작
    MovementComp->UpdateMovement(DeltaTime);
    AttackComp->ProcessAttack(DeltaTime);
    PerceptionComp->ScanEnvironment();
}

Sin/Cos 복합 함수 Movement

저비용 고효율 연산을 이용한 자연스러운 무브먼트 패턴

핵심 코드

void AMoveAction::Execute()
{
    float Time = GetWorld()->GetTimeSeconds();
    
    // 랜덤하게 Sin 또는 Cos 선택
    bool bUseSin = FMath::RandBool();
    FVector NewLoc = Owner->GetActorLocation();
    
    if (bUseSin) {
        // Sin 패턴: 좌우 웨이브
        NewLoc.Y += FMath::Sin(Time * Speed) * Amp;
    } else {
        // Cos 패턴: 상하 웨이브
        NewLoc.X += FMath::Cos(Time * Speed) * Amp;
    }
    
    Owner->SetActorLocation(NewLoc);
}

로프 구속 물리 시스템 (Swing Physics)
⚠️

구현 배경 및 문제

Spider-Man Swing Physics Problem & Solution

왼쪽: 구속 조건 없음 (나선형 이탈) | 오른쪽: 로프 구속 적용 (원형 유지)

문제점

  • 1. 단순히 중력만 적용하면 캐릭터가 앵커 포인트에서 무한히 멀어진다.
  • 2. 로프가 팽팽해진 순간 반지름 방향 속도를 제거하지 않으면 나선형으로 멀어짐.
  • 3. 위치 보정 없이 속도만 수정하면 프레임 간 누적 오차로 로프가 늘어남.

해결 방법

  • ✓ 속도 벡터를 반지름 방향 + 접선 방향으로 분해 (Dot Product 투영).
  • ✓ 로프가 팽팽할 때 반지름 방향 속도 성분 제거 → 접선 방향만 유지.
  • ✓ 위치 보정: CorrectPos = Anchor - RopeDir × RopeLength.
  • ✓ SwingDamping 감쇠 적용으로 자연스러운 에너지 소실 표현.

구현 조건

if (Distance > RopeLength)
{
  // 반지름 방향 속도 제거
  // 위치 보정 적용
}
속도 벡터 분해 및 물리 시뮬레이터

속도 벡터 분해 (Projection)

전체 속도(v)
반지름 성분(v_r) 제거
접선 성분(v_t) 유지

로프가 팽팽한 시점에 Dot Product를 통해 캐릭터의 속도를 두 성분으로 분해합니다. 반지름 방향 성분(v_r)을 강제로 0으로 만들고 접선 성분(v_t)만 유지함으로써 원운동 궤적을 강제합니다.

Constraint-based Swing Physics

Spider-Man Swing

1. 문제의 본질 규명 단순 중력만 적용하면 캐릭터가 앵커 포인트에서 계속 멀어진다. 스윙의 핵심은 낙하가 아니라 로프 길이를 유지하는 구속 조건을 매 프레임 만족시키는 것이다.
2. 전략 고도화 속도를 반지름 방향과 접선 방향으로 분해하고, 로프가 팽팽한 순간 반지름 성분만 제거했다. 여기에 위치 보정을 추가해 프레임 누적 오차로 로프 길이가 늘어나는 현상을 억제했다.
