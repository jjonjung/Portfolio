// Sin·Cos 다주파수 조합으로 불규칙 이동 벡터 생성
void UAiBrainComponent::ChaoticChase()
{
    float t = GetWorld()->GetTimeSeconds();

    float Chaos1 = FMath::Sin(t * 1.3f) * 0.8f;
    float Chaos2 = FMath::Sin(t * 2.7f) * FMath::Cos(t * 0.9f) * 0.5f;
    float Chaos3 = FMath::Sin(t * 3.1f) * FMath::Cos(t * 2.3f) * 0.7f;

    FVector ChaoticDir = BaseDirection
        + (SideDirection                                          * Chaos1)
        + (ToTarget.GetSafeNormal()                            * Chaos2)
        + (FVector::CrossProduct(SideDirection, FVector::UpVector) * Chaos3);

    Character->AddMovementInput(ChaoticDir.GetSafeNormal(), 2.0f);

    if (FMath::Fmod(t, 0.7f) < 0.1f)
        RecievedIntent(FName("Input.Fire"));
}
